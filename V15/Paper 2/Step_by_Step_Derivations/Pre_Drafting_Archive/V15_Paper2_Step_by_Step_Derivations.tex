\section{Protocol 1: Step-Index and Parabolic GRIN RCF Eigenmodes, Tolerances, and Multi-Ring SDM}
\label{sec:p2_rcf}

\begin{targeteq}{Liouville-Transformed RCF Hamiltonian, GRIN Cutoff, and Zero-Forcing Penalties}
\begin{align*}
  -\chi''_l(r) + \left[\frac{l^2 - \frac{1}{4}}{r^2} - k_0^2 n^2(r)\right]\chi_l(r) &= -\beta_l^2\chi_l(r), \qquad u_l(r) = \frac{\chi_l(r)}{\sqrt{r}}, \\
  \mathcal{P}_{\mathrm{corr}} = \frac{1}{N}\operatorname{Tr}(\gamma^{-1}), &\qquad \mathcal{P}_{\mathrm{AWGN}} = \frac{1}{N}\operatorname{Tr}(\gamma^{-2}), \qquad \mathrm{BER} = \frac{1}{2}\operatorname{erfc}\!\left(\sqrt{\frac{\eta_l (E_b/N_0)}{2\mathcal{P}}}\right).
\end{align*}
\end{targeteq}

\subsection{Sturm--Liouville Reduction and Liouville Symmetrization}
Under the weakly guiding scalar Helmholtz equation $(\nabla_\perp^2 + k_0^2 n^2(r))\psi_{\mathrm{RCF},l} = \beta_l^2\psi_{\mathrm{RCF},l}$ with ansatz $\psi_{\mathrm{RCF},l}(r,\phi) = (2\pi)^{-1/2} u_l(r) e^{il\phi}$ ($\int_0^\infty \abs{u_l(r)}^2 r\,\dd r = 1$), substituting $\nabla_\perp^2 = \frac{1}{r}\frac{\partial}{\partial r}(r\frac{\partial}{\partial r}) + \frac{1}{r^2}\frac{\partial^2}{\partial\phi^2}$ yields
\begin{equation}
  -\frac{1}{r}\frac{\dd}{\dd r}\!\left(r\frac{\dd u_l}{\dd r}\right) + \left[\frac{l^2}{r^2} - k_0^2 n^2(r)\right] u_l(r) = -\beta_l^2 u_l(r).
\end{equation}
To obtain a symmetric (self-adjoint in unweighted $L^2(\dd r)$) tridiagonal discretization without first derivatives, apply the Liouville substitution $u_l(r) = r^{-1/2}\chi_l(r)$ (so $\int_0^\infty \abs{\chi_l(r)}^2\dd r = \int_0^\infty \abs{u_l(r)}^2 r\,\dd r = 1$). Compute each derivative explicitly:
\begin{align}
  u'_l &= r^{-1/2}\chi'_l - \frac{1}{2}r^{-3/2}\chi_l, \qquad r u'_l = r^{1/2}\chi'_l - \frac{1}{2}r^{-1/2}\chi_l, \\
  \frac{\dd}{\dd r}(r u'_l) &= r^{1/2}\chi''_l + \frac{1}{2}r^{-1/2}\chi'_l - \frac{1}{2}r^{-1/2}\chi'_l + \frac{1}{4}r^{-3/2}\chi_l = r^{1/2}\left(\chi''_l + \frac{1}{4r^2}\chi_l\right).
\end{align}
Dividing by $r^{1/2}$ converts $-\frac{1}{r}(r u'_l)' + \frac{l^2}{r^2}u_l$ exactly to $-\chi''_l + \frac{l^2 - 1/4}{r^2}\chi_l$:
\begin{equation}
  -\frac{\dd^2\chi_l}{\dd r^2} + \left[\frac{l^2 - \frac{1}{4}}{r^2} - k_0^2 n^2(r)\right]\chi_l(r) = -\beta_l^2 \chi_l(r).
  \label{eq:walk_rcf_sym}
\end{equation}
On a uniform grid $r_j = r_{\min} + j h$ ($j = 0,\dots,N_r-1$), replacing $-\chi''_l(r_j) \approx (-\chi_{j-1} + 2\chi_j - \chi_{j+1})/h^2$ produces a real symmetric tridiagonal matrix whose lowest eigenvalue $\lambda_0 = -\beta_l^2$ yields $n_{\mathrm{eff}}(l) = \sqrt{-\lambda_0}/k_0$.

\subsection{Step-Index vs.\ Parabolic Graded-Index (GRIN) RCF Profiles}
We benchmark two profiles sharing identical ring geometry $r_1 = 6.0\,\mu\mathrm{m}$, $r_2 = 9.0\,\mu\mathrm{m}$ ($r_{\mathrm{mid}} = 7.5\,\mu\mathrm{m}$, $a_{\mathrm{ring}} = 1.5\,\mu\mathrm{m}$), $n_{\mathrm{clad}} = 1.444$, peak $\Delta n = 0.033$ ($n_{\mathrm{core}} = 1.477$), and $\lambda = 1550\,\mathrm{nm}$:
\begin{align}
  n_{\mathrm{step}}(r) &= \begin{cases} n_{\mathrm{core}}, & r \in [r_1, r_2], \\ n_{\mathrm{clad}}, & \text{otherwise}, \end{cases} \\
  n_{\mathrm{GRIN}}^2(r) &= n_{\mathrm{clad}}^2 + \bigl(n_{\mathrm{core}}^2 - n_{\mathrm{clad}}^2\bigr)\max\!\left[0,\; 1 - \left(\frac{r - r_{\mathrm{mid}}}{a_{\mathrm{ring}}}\right)^2\right].
\end{align}
The integrated index-volume of the parabolic ring satisfies $\int_{r_1}^{r_2} (1 - u^2)\,\dd u = \frac{2}{3}\int_{r_1}^{r_2} 1\,\dd u$, reducing effective V-number by $\sqrt{2/3} \approx 0.816$. Consequently, whereas step-index supports $l = 0\text{--}8$ ($n_{\mathrm{eff}}(8) = 1.44443 > n_{\mathrm{clad}}$), the parabolic GRIN RCF reaches cladding cutoff precisely past $l = 6$: $n_{\mathrm{eff}}^{\mathrm{GRIN}}(6) = 1.44760 > 1.444$, while $n_{\mathrm{eff}}^{\mathrm{GRIN}}(7) = 1.44295 < 1.444$ --- explaining from first principles the $l = 1\text{--}6$ guided window observed experimentally in Villalba \emph{et al.}!

For $N$ concentric radial PVB channels $\{\psi_{l,R_n}\}_{n=1}^N$ carrying data symbols $\mathbf{s} = (s_1,\dots,s_N)^{\mathsf{T}}$ ($\mathbb{E}[\mathbf{s}\mathbf{s}^\dagger] = P_s\mathbb{I}_N$), matched-filter projection onto $\{\psi_{l,R_m}\}$ yields the noisy vector $\mathbf{y} = \gamma\mathbf{s} + \boldsymbol{\varepsilon}$, where $[\gamma]_{mn} = \braket{\psi_{l,R_m}}{\psi_{l,R_n}}$. Applying the linear zero-forcing (ZF) equalizer $\hat{\mathbf{s}} = \gamma^{-1}\mathbf{y} = \mathbf{s} + \gamma^{-1}\boldsymbol{\varepsilon}$ produces post-equalization noise covariance $\Sigma_{\mathrm{ZF}} = \gamma^{-1}\mathbb{E}[\boldsymbol{\varepsilon}\boldsymbol{\varepsilon}^\dagger]\gamma^{-1}$ (since $\gamma = \gamma^\dagger$):
\begin{align}
  \text{Correlated optical noise }\mathbb{E}[\boldsymbol{\varepsilon}\boldsymbol{\varepsilon}^\dagger] = \sigma^2\gamma: &\quad \Sigma_{\mathrm{ZF}} = \sigma^2\gamma^{-1} \implies \mathcal{P}_{\mathrm{corr}} = \frac{\operatorname{Tr}(\Sigma_{\mathrm{ZF}})}{N\sigma^2} = \frac{1}{N}\operatorname{Tr}(\gamma^{-1}), \\
  \text{Uncorrelated receiver AWGN }\mathbb{E}[\boldsymbol{\varepsilon}\boldsymbol{\varepsilon}^\dagger] = \sigma^2\mathbb{I}_N: &\quad \Sigma_{\mathrm{ZF}} = \sigma^2\gamma^{-2} \implies \mathcal{P}_{\mathrm{AWGN}} = \frac{\operatorname{Tr}(\Sigma_{\mathrm{ZF}})}{N\sigma^2} = \frac{1}{N}\operatorname{Tr}(\gamma^{-2}).
\end{align}

\subsection{Orthogonal Subspace Projection via the Inverse Gram Matrix}
Let $\mathcal{V}_N = \operatorname{span}\{v_n\}_{n=1}^N \subset \mathcal{H}_{l_0}$ be any $N$-dimensional linearly independent subspace (either the propagated discrete PVB states $v_n = \mathcal{U}_z\psi_{l_0,R_n}$ or the matched-dimension orthonormal $\mathrm{LG}_{p,l_0}$ modes $v_n = \mathcal{U}_z\psi_{n-1,l_0}$). Note that because $\mathcal{U}_z$ is unitary, the propagated Gram matrix is identical to the waist Gram matrix: $[\gamma]_{mn} = \braket{\mathcal{U}_z\psi_{l_0,R_m}}{\mathcal{U}_z\psi_{l_0,R_n}} = \braket{\psi_{l_0,R_m}}{\psi_{l_0,R_n}}$.

We seek the unique element $\hat{\psi} = \sum_{n=1}^N c_n v_n \in \mathcal{V}_N$ minimizing $\norm{\psi_{\mathrm{turb}} - \hat{\psi}}_{L^2}^2$. By the Hilbert projection theorem, the residual $\psi_{\mathrm{turb}} - \hat{\psi}$ must be orthogonal to every $v_m$:
\begin{equation}
  \braket{v_m}{\psi_{\mathrm{turb}} - \sum_{n=1}^N c_n v_n} = 0 \iff \sum_{n=1}^N \gamma_{mn} c_n = \braket{v_m}{\psi_{\mathrm{turb}}} \iff c_n = \sum_{m=1}^N (\gamma^{-1})_{nm}\braket{v_m}{\psi_{\mathrm{turb}}}.
\end{equation}
Substituting $c_n$ gives $P_{\mathcal{V}_N}\psi_{\mathrm{turb}} = \sum_{n,m=1}^N (\gamma^{-1})_{nm}\braket{v_m}{\psi_{\mathrm{turb}}}v_n$. Checking idempotence and self-adjointness explicitly:
\begin{align}
  P_{\mathcal{V}_N}^2\psi &= \sum_{n,m,j,k} (\gamma^{-1})_{nm}(\gamma^{-1})_{jk}\braket{v_m}{v_j}\braket{v_k}{\psi} v_n = \sum_{n,k} \Bigl(\sum_{m,j} (\gamma^{-1})_{nm}\gamma_{mj}(\gamma^{-1})_{jk}\Bigr)\braket{v_k}{\psi}v_n = P_{\mathcal{V}_N}\psi, \\
  \braket{\phi}{P_{\mathcal{V}_N}\psi} &= \sum_{n,m} (\gamma^{-1})_{nm}\braket{\phi}{v_n}\braket{v_m}{\psi} = \sum_{n,m} \bigl[(\gamma^{-1})_{mn}\braket{v_n}{\phi}\bigr]^*\braket{v_m}{\psi} = \braket{P_{\mathcal{V}_N}\phi}{\psi},
\end{align}
since $\gamma^{-1}$ is Hermitian. Because every basis vector $v_n$ carries pure azimuthal dependence $e^{il_0\phi}$, $P_{\mathrm{PVB},l_0}$ annihilates every $l \neq l_0$ azimuthal crosstalk component identically!

\subsection{Algebraic Proof of the Single-Plane Rank-$(2d-1)$ Ceiling}
On $\mathcal{H}_d^{(l)} = \operatorname{span}\{\ket{\psi_{p,l}}\}_{p=0}^{d-1}$, any Hermitian operator $\rho$ has $d^2$ real parameters: $d$ real diagonals, $d(d-1)/2$ real-symmetric off-diagonals, and $d(d-1)/2$ imaginary-antisymmetric off-diagonals. At a single waist plane ($z=0$), the PVB projection coefficients $c_{l,p}(R) = \braket{\Phi_{l,R}}{\psi_{p,l}} \in \mathbb{R}$ are strictly real. Hence every matrix element $[\ket{\Phi_{l,R}}\bra{\Phi_{l,R}}]_{pp'} = c_{l,p}(R) c_{l,p'}(R)$ is real and symmetric, annihilating all $d(d-1)/2$ antisymmetric imaginary coherences identically.

Even within the $d(d+1)/2$-dimensional real-symmetric subspace, recall from V15 Paper~1 that $c_{l,p}(R) = \mathcal{A}_{0,l}(R) Q_p(\zeta_R)$ where $\mathcal{A}_{0,l}(R)$ is a $p$-independent envelope and $Q_p(\zeta_R)$ is a polynomial of exact degree $p$ in $\zeta_R = R^2 w_0^2/[w^2(w_0^2+w^2)]$. Therefore every product
\begin{equation}
  c_{l,p}(R) c_{l,p'}(R) = \bigl[\mathcal{A}_{0,l}(R)\bigr]^2 Q_p(\zeta_R) Q_{p'}(\zeta_R)
\end{equation}
is $\mathcal{A}_{0,l}^2(R)$ times a polynomial in $\zeta_R$ of degree $p + p' \le 2(d-1) = 2d - 2$. Since the space of polynomials of degree $\le 2d-2$ has dimension $(2d-2) + 1 = 2d - 1$, the single-plane POVM elements span \emph{at most} $2d-1$ linearly independent operators! At $d = 4$, $2d - 1 = \mathbf{7}$ (matching numerical SVD rank $7/16$, strictly below $d(d+1)/2 = 10$).

\subsection{Passive Gouy Phase Lifting, POVM Completion, and Closed-Form MUB Bound}
Splitting the photon passively across $M \ge d+1$ propagation planes $z_m$ with wavefront-compensated rescaled analyzers $\phi_{m,k}(r,\phi) = \mathbf{\sigma_m} e^{+ik_0 r^2/[2R_c(z_m)]}\Phi_{l,\sigma_m R_k}(r,\phi;\,\sigma_m w)$ (using the exact $\sigma_m^{+1}$ scaling proven in Sec.~4 of the V15 Paper~1 Walkthrough!) introduces the differential Gouy phase factor:
\begin{equation}
  [\Pi_{m,k}]_{p,p'} = \frac{R_k\Delta R}{M}\,e^{-i 2(p - p')\vartheta_m}\,c_{l,p}(R_k) c_{l,p'}(R_k).
\end{equation}
The distinct Fourier harmonics $e^{-i 2(p-p')\vartheta_m}$ decouple all off-diagonal pairs $(p,p')$ with $p \neq p'$ and restore imaginary coherences cleanly for $M \ge d+1$, lifting rank from $7/16$ to full IC rank $\mathbf{16/16}$. Because the Riemann bin sum approximates $\frac{1}{M}\sum_{m=1}^M P_{\mathcal{H}_d} S_w(z_m) P_{\mathcal{H}_d} \le \mathbb{I}_d$ (by Paper~1's Bessel upper bound $B = 1$), $\lambda_{\max}(\sum_{m,k}\Pi_{m,k}) = 0.90498 < 1$, guaranteeing positive semidefiniteness of $\Pi_{\mathrm{res}} = \mathbb{I}_d - \sum_{m,k}\Pi_{m,k} \ge 0$.

Parameterizing $\rho(\boldsymbol{\theta}) = \mathbb{I}_d/d + \sum_{j=1}^{d^2-1}\theta_j B_j$ in an orthonormal traceless Gell-Mann basis ($\operatorname{Tr}(B_j B_k) = \delta_{jk}$), the outcome probability is $p_\alpha(\boldsymbol{\theta}) = \operatorname{Tr}(\rho\Pi_\alpha) = \frac{1}{d}\operatorname{Tr}(\Pi_\alpha) + \sum_j \theta_j \operatorname{Tr}(\Pi_\alpha B_j)$, so $\partial_{\theta_j} p_\alpha = \operatorname{Tr}(\Pi_\alpha B_j)$. The classical Fisher information matrix at $\boldsymbol{\theta} = \mathbf{0}$ is
\begin{equation}
  F_{jk} = \sum_\alpha \frac{(\partial_{\theta_j} p_\alpha)(\partial_{\theta_k} p_\alpha)}{p_\alpha(\mathbf{0})} = \sum_\alpha \frac{\operatorname{Tr}(\Pi_\alpha B_j)\operatorname{Tr}(\Pi_\alpha B_k)}{\operatorname{Tr}(\Pi_\alpha)/d}.
\end{equation}
For $d+1$ mutually unbiased bases (MUBs) $\{\ket{e_{b,r}}\}_{r=1}^d$ ($b=1,\dots,d+1$) with uniform POVM $\Pi_{b,r} = \frac{1}{d+1}\ket{e_{b,r}}\bra{e_{b,r}}$, we have $p_{b,r}(\mathbf{0}) = \frac{1}{d(d+1)}$. Choosing $\{B_j\}$ aligned with the $d+1$ mutually orthogonal $(d-1)$-dimensional traceless diagonal blocks of each MUB yields $\sum_{r=1}^d \bra{e_{b,r}}B_j\ket{e_{b,r}}\bra{e_{b,r}}B_k\ket{e_{b,r}} = \operatorname{Tr}(B_j B_k) = \delta_{jk}$ when $B_j, B_k$ belong to block $b$ (and $0$ otherwise because MUB off-diagonals are unbiased). Hence
\begin{equation}
  [\mathbf{F}_{\mathrm{MUB}}]_{jk} = \frac{1/(d+1)^2}{1/[d(d+1)]}\delta_{jk} = \frac{d}{d+1}\delta_{jk} \implies \operatorname{Tr}\!\left(\mathbf{F}_{\mathrm{MUB}}^{-1}\right) = (d^2 - 1)\frac{d+1}{d}.
\end{equation}
At $d=4$, $\operatorname{Tr}(\mathbf{F}_{\mathrm{MUB}}^{-1}) = 15\times 5/4 = 18.75$, compared to $\operatorname{Tr}(\mathbf{F}_{\mathrm{PVB}}^{-1}) = 41.80$ ($2.23\times$ factor).

\begin{pencheck}[Verified Extended IC-POVM Results: Pure, Haar, and Axial Misalignment (Suite 3)]
For coherent superposition $\ket{\psi_{\mathrm{coh}}} = (\ket{0,1}+i\ket{3,1})/\sqrt{2}$, $R\hat{\rho}R$ MLE achieves $F = 0.9398$ ($300$ photons), $0.9738$ ($3{,}000$), $\mathbf{0.9823}$ ($10{,}000$), and $0.9870$ ($30{,}000$). \end{pencheck}

\section{Protocol 4: Circular-Iris Bell--CHSH Asymptotics and Detection-Loophole Threshold}
\label{sec:p2_chsh}

\begin{targeteq}{Incomplete Gamma Law, Sixth-Order Tsirelson Bound, and Loophole-Free Iris Radius}
\begin{align*}
  t_l^{\mathrm{LG}}(R_{\mathrm{iris}}) &= P\!\left(\abs{l}+1, \frac{2R_{\mathrm{iris}}^2}{w_0^2}\right), \qquad
  t_l^{\mathrm{PVB}}(R_0) = \frac{1}{2} + c_0\frac{w}{R_0} - \frac{4l^2 - 1}{16\sqrt{2\pi}}\left(\frac{w}{R_0}\right)^3 + \mathcal{O}\!\left(\frac{w^5}{R_0^5}\right), \\
  2\sqrt{2} - S_{\mathrm{PVB}}^{(2)}(R_0) &\approx 2\sqrt{2}\,\delta_{14}^2 = \mathcal{O}\!\left((l_1^2-l_2^2)^2\left(\frac{w}{R_0}\right)^6\right), \qquad \eta_{\mathrm{crit}}^{\mathrm{CHSH}} = \frac{2}{1+\sqrt{2}} = 82.8427\%.
\end{align*}
\end{targeteq}

\subsection{Exact $\mathrm{LG}_{0,l}$ Iris Transmission}
For $\psi_{0,l}(r,\phi) = \sqrt{\frac{2}{\pi\abs{l}!}}\frac{1}{w_0}(\frac{\sqrt{2}r}{w_0})^{\abs{l}} e^{-r^2/w_0^2 + il\phi}$, integrating $\abs{\psi_{0,l}}^2$ over $r \in [0, R_{\mathrm{iris}}]$ with $u = 2r^2/w_0^2$ ($\dd u = 4r\,\dd r/w_0^2 \implies 2\pi r\,\dd r = \frac{\pi w_0^2}{2}\dd u$):
\begin{equation}
  t_l^{\mathrm{LG}}(R_{\mathrm{iris}}) = \frac{2}{\pi\abs{l}! w_0^2}\cdot\frac{\pi w_0^2}{2}\int_0^{2R_{\mathrm{iris}}^2/w_0^2} u^{\abs{l}} e^{-u}\,\dd u = \frac{\gamma(\abs{l}+1, 2R_{\mathrm{iris}}^2/w_0^2)}{\Gamma(\abs{l}+1)} = P\!\left(\abs{l}+1,\;\frac{2R_{\mathrm{iris}}^2}{w_0^2}\right).
\end{equation}

\subsection{Step-by-Step Hankel Expansion of $t_l^{\mathrm{PVB}}(R_0)$ to Order $(w/R_0)^3$}
Set $\epsilon \equiv w/R_0 \ll 1$, $\mu \equiv 4l^2 - 1$, and shift to the dimensionless ring coordinate $r = R_0(1 + \epsilon u)$ ($r\,\dd r = R_0^2\epsilon(1+\epsilon u)\dd u$). Inside the iris integral, the unnormalized intensity density is proportional to $r e^{-2u^2} [e^{-2R_0 r/w^2} I_{\abs{l}}(2R_0 r/w^2)]^2$. Using $x = \frac{2R_0 r}{w^2} = \frac{2}{\epsilon^2}(1+\epsilon u)$ and Hankel's large-argument expansion $e^{-x} I_{\abs{l}}(x) = (2\pi x)^{-1/2}[1 - \mu/(8x) + \mathcal{O}(x^{-2})]$:
\begin{equation}
  r\,\bigl[e^{-x} I_{\abs{l}}(x)\bigr]^2 \propto \frac{r}{x}\left(1 - \frac{\mu}{8x} + \mathcal{O}(\epsilon^4)\right)^2 \propto \frac{1+\epsilon u}{1+\epsilon u}\left[1 - \frac{\mu\epsilon^2}{8(1+\epsilon u)} + \mathcal{O}(\epsilon^4)\right] = 1 - \frac{\mu\epsilon^2}{8} + \frac{\mu\epsilon^3}{8} u + \mathcal{O}(\epsilon^4).
\end{equation}
Observe the two crucial cancellations! First, $r$ in the polar Jacobian cancels $1/r$ from $1/x$ \emph{identically}---leaving a pure even Gaussian weight $e^{-2u^2}$ at $\mathcal{O}(\epsilon^0)$ and $\mathcal{O}(\epsilon^2)$! Second, the constant shift $-\mu\epsilon^2/8$ cancels completely between the numerator $\int_{-\infty}^0$ and denominator $\int_{-\infty}^\infty$, leaving the first $l$-dependent contribution strictly at $\mathcal{O}(\epsilon^3)$ via the odd moment $\int_{-\infty}^0 u\,e^{-2u^2}\,\dd u$! Compute the two Gaussian moments explicitly:
\begin{equation}
  \int_{-\infty}^\infty e^{-2u^2}\,\dd u = \sqrt{\frac{\pi}{2}}, \qquad
  \int_{-\infty}^0 u\,e^{-2u^2}\,\dd u = \left[-\frac{1}{4}e^{-2u^2}\right]_{-\infty}^0 = -\frac{1}{4}.
\end{equation}
Therefore, the $l$-dependent term in $t_l^{\mathrm{PVB}}(R_0) = \frac{\int_{-\infty}^0 e^{-2u^2}(1 + \frac{\mu\epsilon^3}{8}u + \dots)\dd u}{\int_{-\infty}^\infty e^{-2u^2}(1 + \dots)\dd u}$ is
\begin{equation}
  \Delta t_l^{(3)} = \frac{\mu\epsilon^3}{8}\cdot\frac{-1/4}{\sqrt{\pi/2}} = -\frac{\mu\epsilon^3}{32}\sqrt{\frac{2}{\pi}} = -\frac{4l^2 - 1}{16\sqrt{2\pi}}\left(\frac{w}{R_0}\right)^3!
  \label{eq:walk_t_pvb_3}
\end{equation}
For $l_1 = 1$ ($\mu_1 = 3$) and $l_2 = 4$ ($\mu_2 = 63$), $\mu_2 - \mu_1 = 60$, giving the relative transmission imbalance
\begin{equation}
  \delta_{14} \equiv \frac{t_1 - t_4}{t_1 + t_4} \approx \frac{t_1 - t_4}{1} = \frac{60}{16\sqrt{2\pi}}\left(\frac{w}{R_0}\right)^3 = \frac{15}{4\sqrt{2\pi}}\left(\frac{w}{R_0}\right)^3 = 1.4960\times 10^{-3}\quad\text{at }w/R_0 = 0.1
\end{equation}
(matching exact quadrature $1.510\times 10^{-3}$ to $0.9\%$).

\subsection{Horodecki CHSH Maximum and Symmetric Detection-Loophole Threshold}
Behind single-arm or symmetric two-arm circular irises, the post-selected pure entangled state is $\ket{\Psi_{\mathrm{ps}}} = \frac{\sqrt{p_1}\ket{00} + \sqrt{p_2}\ket{11}}{\sqrt{p_1+p_2}}$ ($p_l = t_l$ one-arm, $p_l = t_l^2$ two-arm). Computing its Pauli correlation matrix $T_{ij} = \bra{\Psi_{\mathrm{ps}}}\sigma_i\otimes\sigma_j\ket{\Psi_{\mathrm{ps}}}$ gives $T = \operatorname{diag}(\mathcal{V}, -\mathcal{V}, 1)$ with visibility $\mathcal{V} = \frac{2\sqrt{p_1 p_2}}{p_1+p_2}$. By the Horodecki theorem, $S_{\max} = 2\sqrt{\lambda_1(T^{\mathsf{T}}T) + \lambda_2(T^{\mathsf{T}}T)} = 2\sqrt{1 + \mathcal{V}^2}$. Writing $t_1/t_4 = (1+\delta_{14})/(1-\delta_{14})$:
\begin{align}
  \text{One-arm: } &\quad \mathcal{V}^{(1)} = \sqrt{1 - \delta_{14}^2} \implies S^{(1)} = 2\sqrt{2 - \delta_{14}^2} = 2\sqrt{2}\left(1 - \frac{\delta_{14}^2}{4} + \mathcal{O}(\delta_{14}^4)\right) \implies 2\sqrt{2} - S^{(1)} \approx \frac{\delta_{14}^2}{\sqrt{2}} = 1.58\times 10^{-6}, \\
  \text{Two-arm: } &\quad \mathcal{V}^{(2)} = \frac{1 - \delta_{14}^2}{1 + \delta_{14}^2} \approx 1 - 2\delta_{14}^2 \implies 2\sqrt{2} - S^{(2)} \approx 2\sqrt{2}\,\delta_{14}^2 = 6.33\times 10^{-6}.
\end{align}
Finally, to test local realism \emph{without} the fair-sampling post-selection assumption using a maximally entangled two-qubit state and symmetric single-arm efficiency $\eta$, assigning undetected events to outcome $+1$ bounds the unconditional CHSH correlation by $S_{\mathrm{uncond}}(\eta) = \eta^2(2\sqrt{2}) + 2(1-\eta)^2$ (equivalently via Clauser--Horne, $\eta S_{\mathrm{Bell}} - 2(1-\eta) > 0$). Requiring $S_{\mathrm{uncond}}(\eta) > 2$:
\begin{equation}
  2\sqrt{2}\,\eta^2 + 2(1 - 2\eta + \eta^2) > 2 \iff (2\sqrt{2} + 2)\eta^2 - 4\eta > 0 \iff \eta > \frac{2}{1+\sqrt{2}} = 2(\sqrt{2}-1) = 0.828427.
\end{equation}
